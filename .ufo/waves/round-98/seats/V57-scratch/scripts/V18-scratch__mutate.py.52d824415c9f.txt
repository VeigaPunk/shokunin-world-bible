#!/usr/bin/env python3
"""V18 round-98 preservation-mutation harness: build mutated copies of the delivered
r98-clerk-filing fixture. Writes ONLY under V18-scratch/. Never touches corpus/.

Each mutant is a full copy of corpus/fixtures/r98-clerk-filing with one preservation
violation applied. Prints a manifest of the exact edit so the report can quote it.
"""

import json
import os
import re
import shutil
import sys
from pathlib import Path

REPO = Path("/home/vgpnk/Projects/shokunin/lore-shokunin")
SCRATCH = REPO / ".ufo/waves/round-98/seats/V18-scratch"
SRC = Path(os.environ.get("V18_SRC", str(REPO / "corpus/fixtures/r98-clerk-filing")))
MUTANTS = Path(os.environ.get("V18_MUTANTS", str(SCRATCH / "mutants")))
MAP = "claim-excerpt-map.md"


def load(path):
    return path.read_text(encoding="utf-8")


def save(path, text):
    path.write_text(text, encoding="utf-8")


def fresh(name):
    target = MUTANTS / name
    if target.exists():
        shutil.rmtree(target)
    shutil.copytree(SRC, target)
    return target


def data_row_lines(lines):
    return [i for i, line in enumerate(lines) if re.match(r"^\|\s*J[0-9]+\s*\|", line)]


def last_row(lines):
    return data_row_lines(lines)[-1]


def row_cells(line):
    return line.strip().strip("|").split("|")


def rebuild_row(cells):
    return "|" + "|".join(cells) + "|"


def header_cells(lines):
    for j, line in enumerate(lines):
        if re.match(r"^\|\s*ID\s*\|", line):
            return [c.strip().strip("`").lower() for c in row_cells(line)]
    raise SystemExit("no header row found")


def set_declaration(lines, facts=None, editorial=None, total=None, anchors=None):
    for i, line in enumerate(lines):
        if line.startswith("**Expected inventory:**"):
            new = line
            if facts is not None:
                new = re.sub(r"([0-9]+) JSON facts", f"{facts} JSON facts", new)
            if editorial is not None:
                new = re.sub(r"([0-9]+) editorial rows?", f"{editorial} editorial rows", new)
            if total is not None:
                new = re.sub(r"([0-9]+) total mapped rows", f"{total} total mapped rows", new)
            if anchors is not None:
                new = re.sub(r"([0-9]+) distinct excerpt anchors", f"{anchors} distinct excerpt anchors", new)
            lines[i] = new
            return new
    raise SystemExit("declaration line not found")


def row_index_for_pointer(lines, pointer):
    for i in data_row_lines(lines):
        cells = row_cells(lines[i])
        if pointer in cells[1]:
            return i
    raise SystemExit(f"no map row for pointer {pointer}")


def manifest(name, note, detail):
    print(f"MUTANT {name}: {note}")
    print(f"  {detail}")


def m1_span_break():
    t = fresh("m1-span-break")
    p = t / MAP
    lines = load(p).splitlines()
    i = last_row(lines)
    cells = row_cells(lines[i])
    header = header_cells(lines)
    span_index = next(k for k, h in enumerate(header) if h == "line span")
    old = cells[span_index].strip()
    cells[span_index] = " 1 "
    lines[i] = rebuild_row(cells)
    save(p, "\n".join(lines) + "\n")
    manifest("m1-span-break", f"last JSON-facts row span {old} -> 1 (anchor now outside span)",
             lines[i][:160])


def m2_row_drop():
    t = fresh("m2-row-drop")
    p = t / MAP
    lines = load(p).splitlines()
    i = last_row(lines)
    dropped = lines.pop(i)
    save(p, "\n".join(lines) + "\n")
    manifest("m2-row-drop", "last JSON-facts row deleted; declaration untouched", dropped[:160])


def m2b_row_drop_declared():
    t = fresh("m2-row-drop-declared")
    p = t / MAP
    lines = load(p).splitlines()
    n_facts = len(data_row_lines(lines)) - 1
    i = last_row(lines)
    lines.pop(i)
    new = set_declaration(lines, facts=n_facts, total=n_facts + 2)
    save(p, "\n".join(lines) + "\n")
    manifest("m2b-row-drop-declared", f"last row deleted + declaration downgraded to {n_facts}", new)


def rename_last_entity_claims(t):
    p = t / "entities.json"
    data = json.loads(load(p))
    entity = data[-1]
    old_key = "claims"
    new_key = "assertions"
    entity[new_key] = entity.pop(old_key)
    save(p, json.dumps(data, indent=2) + "\n")
    return entity["id"], len(data) - 1


def m3a_claims_rename_keep_row():
    t = fresh("m3a-claims-rename-keep-row")
    entity_id, idx = rename_last_entity_claims(t)
    manifest("m3a-claims-rename-keep-row",
             f"entities.json last entity {entity_id!r} key claims -> assertions; map row kept",
             f"entities.json#/{idx}/claims/0 row now phantom")


def m3b_claims_rename_row_removed():
    t = fresh("m3b-claims-rename-row-removed")
    entity_id, idx = rename_last_entity_claims(t)
    p = t / MAP
    lines = load(p).splitlines()
    i = row_index_for_pointer(lines, f"entities.json#/{idx}/claims/0")
    dropped = lines.pop(i)
    save(p, "\n".join(lines) + "\n")
    manifest("m3b-claims-rename-row-removed",
             f"entities.json last entity {entity_id!r} key claims -> assertions + map row removed; declaration untouched",
             dropped[:160])


def m4_claims_row_deletion():
    t = fresh("m4-claims-row-deletion")
    p = t / "entities.json"
    data = json.loads(load(p))
    entity = data[-1]
    idx = len(data) - 1
    removed = entity.pop("claims")
    save(p, json.dumps(data, indent=2) + "\n")
    m = t / MAP
    lines = load(m).splitlines()
    i = row_index_for_pointer(lines, f"entities.json#/{idx}/claims/0")
    dropped = lines.pop(i)
    save(m, "\n".join(lines) + "\n")
    manifest("m4-claims-row-deletion",
             f"entities.json last entity {entity['id']!r} 'claims' key deleted + its map row deleted; declaration untouched",
             f"{len(removed)} claim(s) removed; {dropped[:140]}")


def last_event_id(t):
    data = json.loads(load(t / "timeline.json"))
    return data[-1]["id"], len(data) - 1


def m5_deps_reduction():
    t = fresh("m5-deps-reduction")
    p = t / "timeline.json"
    data = json.loads(load(p))
    target = data[-1]
    before = list(target.get("depends_on", []))
    target["depends_on"] = []
    save(p, json.dumps(data, indent=2) + "\n")
    manifest("m5-deps-reduction",
             f"timeline.json record {target['id']} depends_on {before} -> []",
             json.dumps(target, ensure_ascii=False)[:160])


def m5b_event_deletion_fully_declared():
    t = fresh("m5b-event-deletion-fully-declared")
    p = t / "timeline.json"
    data = json.loads(load(p))
    removed = data.pop()
    save(p, json.dumps(data, indent=2) + "\n")
    m = t / MAP
    lines = load(m).splitlines()
    i = row_index_for_pointer(lines, f"timeline.json#/{len(data)}")
    lines.pop(i)
    n_facts = len(data_row_lines(lines))
    new = set_declaration(lines, facts=n_facts, total=n_facts + 2, anchors=16)
    save(m, "\n".join(lines) + "\n")
    manifest("m5b-event-deletion-fully-declared",
             f"timeline.json record {removed['id']} deleted + map row deleted + declaration downgraded (self-consistent)",
             new)


def m6_digest_tamper():
    t = fresh("m6-digest-tamper")
    p = t / "sources.json"
    text = load(p)
    save(p, text.replace("fictional", "Fictional", 1))
    manifest("m6-digest-tamper", "sources.json title-case-only byte change",
             "one 'fictional' -> 'Fictional' replacement")


def main():
    if not SRC.is_dir():
        raise SystemExit(f"source fixture missing: {SRC}")
    MUTANTS.mkdir(parents=True, exist_ok=True)
    # structure report for the record
    tl = json.loads(load(SRC / "timeline.json"))
    en = json.loads(load(SRC / "entities.json"))
    so = json.loads(load(SRC / "sources.json"))
    rel = json.loads(load(SRC / "relations.json"))
    lines = load(SRC / MAP).splitlines()
    print(f"SOURCE {SRC.name}: entities={len(en)} relations={len(rel)} events={len(tl)} sources={len(so)}")
    print(f"  last event id={tl[-1]['id']} deps={tl[-1]['depends_on']}")
    print(f"  last entity id={en[-1]['id']} claims={len(en[-1].get('claims', []))}")
    print(f"  map J-rows={len(data_row_lines(lines))}")
    for line in lines:
        if line.startswith("**Expected inventory:**"):
            print(f"  declaration: {line}")
    m1_span_break()
    m2_row_drop()
    m2b_row_drop_declared()
    m3a_claims_rename_keep_row()
    m3b_claims_rename_row_removed()
    m4_claims_row_deletion()
    m5_deps_reduction()
    m5b_event_deletion_fully_declared()
    m6_digest_tamper()
    print("MUTANTS READY under", MUTANTS)


if __name__ == "__main__":
    sys.exit(main())
