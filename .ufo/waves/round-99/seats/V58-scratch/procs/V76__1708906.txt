/bin/bash
-c
cd '/home/vgpnk/Projects/shokunin/lore-shokunin' && SCRATCH=/home/vgpnk/Projects/shokunin/lore-shokunin/.ufo/waves/round-99/seats/V76-scratch
LOG=$SCRATCH/poll2.log
ROOT=/home/vgpnk/Projects/shokunin/lore-shokunin
echo "=== poll2 start $(date -Iseconds) ===" >> "$LOG"
for i in $(seq 1 36); do
  ts=$(date -Iseconds)
  py_sha=$(sha256sum "$ROOT/corpus/tools/dependency_freeze.py" | awk '{print $1}')
  ct_sha=$(sha256sum "$ROOT/corpus/tools/dependency-freeze-contract.json" | awk '{print $1}')
  py_sz=$(stat -c %s "$ROOT/corpus/tools/dependency_freeze.py")
  ct_sz=$(stat -c %s "$ROOT/corpus/tools/dependency-freeze-contract.json")
  s1=$(ls -1 "$ROOT/.ufo/waves/round-99/seats"/S1*.md 2>/dev/null | tr '\n' ' ')
  sreports=$(ls -1 "$ROOT/.ufo/waves/round-99/seats"/S*.md 2>/dev/null | xargs -n1 basename 2>/dev/null | tr '\n' ' ')
  markers=$(python3 - << 'PY'
import json
from pathlib import Path
p = Path("/home/vgpnk/Projects/shokunin/lore-shokunin/corpus/tools/dependency-freeze-contract.json")
try:
    c = json.loads(p.read_text())
except Exception as e:
    print(f"UNREADABLE {type(e).__name__}")
    raise SystemExit
pins = c.get("pins") if isinstance(c.get("pins"), dict) else {}
exp = (c.get("self_test") or {}).get("expected") if isinstance(c.get("self_test"), dict) else {}
nfiles = []
schema_ok = 0
for name, pin in pins.items():
    files = pin.get("files") if isinstance(pin, dict) else {}
    nfiles.append(len(files) if isinstance(files, dict) else -1)
    if isinstance(pin, dict) and pin.get("payload_schema_version") == 2 and isinstance(files, dict) and len(files) == 10:
        schema_ok += 1
print(f"pins={len(pins)} schema2_full={schema_ok} file_counts={sorted(set(nfiles))} self_test_n={len(exp) if isinstance(exp, dict) else 0} r99={'r99-window-lapse' in pins} md_case={'markdown_tamper' in exp if isinstance(exp, dict) else False}")
PY
)
  echo "try=$i ts=$ts py=$py_sha sz=$py_sz ct=$ct_sha ctsz=$ct_sz s1=[$s1] reports=[$sreports] $markers" >> "$LOG"
  if [ -n "$s1" ] && echo "$markers" | grep -q 'schema2_full=6'; then
    # stability: hash again after 2s
    sleep 2
    py2=$(sha256sum "$ROOT/corpus/tools/dependency_freeze.py" | awk '{print $1}')
    ct2=$(sha256sum "$ROOT/corpus/tools/dependency-freeze-contract.json" | awk '{print $1}')
    if [ "$py2" = "$py_sha" ] && [ "$ct2" = "$ct_sha" ]; then
      echo "READY try=$i py=$py2 ct=$ct2" >> "$LOG"
      echo READY
      exit 0
    fi
    echo "  unstable py2=$py2 ct2=$ct2" >> "$LOG"
  fi
  sleep 20
done
echo EXHAUSTED >> "$LOG"
echo EXHAUSTED
exit 0
