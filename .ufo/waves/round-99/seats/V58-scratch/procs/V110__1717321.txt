/bin/bash
-c
cd '/home/vgpnk/Projects/shokunin/lore-shokunin' && # bounded multi-poll: 6 tries * 20s, exit early if S1 report + schema v2 pins present
for i in 3 4 5 6 7 8; do
  ready=0
  if ls .ufo/waves/round-99/seats/S1*.md >/dev/null 2>&1; then
    if python3 -c "
import json,sys
c=json.load(open('corpus/tools/dependency-freeze-contract.json'))
pins=c['pins']
ok=all(p.get('payload_schema_version')==2 and len(p.get('files',{}))==10 for p in pins.values())
print('pins', len(pins), 'schema', c.get('schema_version'), 'all_v2_10', ok)
sys.exit(0 if ok else 1)
"; then
      ready=1
    fi
  fi
  echo "poll $i $(date -Is) ready=$ready" | tee -a .ufo/waves/round-99/seats/V110-scratch/poll.log
  ls -1 .ufo/waves/round-99/seats/S*.md 2>/dev/null | tee -a .ufo/waves/round-99/seats/V110-scratch/poll.log
  stat -c '%y %n' corpus/tools/dependency_freeze.py corpus/tools/dependency-freeze-contract.json | tee -a .ufo/waves/round-99/seats/V110-scratch/poll.log
  if [ "$ready" = 1 ]; then
    echo "READY at poll $i" | tee -a .ufo/waves/round-99/seats/V110-scratch/poll.log
    exit 0
  fi
  sleep 20
done
echo "NOT_READY after poll 8" | tee -a .ufo/waves/round-99/seats/V110-scratch/poll.log
exit 0
