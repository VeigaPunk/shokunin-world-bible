import os, time
root = "/home/vgpnk/Projects/shokunin/lore-shokunin"
seats = os.path.join(root, ".ufo/waves/round-99/seats")
log = os.path.join(seats, "V84-scratch/poll.log")
know = os.path.join(root, "corpus/knowledge")
for i in range(8, 18):
    time.sleep(20)
    names = sorted(os.listdir(seats)) if os.path.isdir(seats) else []
    s5 = [n for n in names if n.startswith("S5")]
    r99 = [n for n in os.listdir(know) if "r99" in n] if os.path.isdir(know) else []
    recent = []
    now = time.time()
    if os.path.isdir(know):
        for n in os.listdir(know):
            p = os.path.join(know, n)
            if os.path.isfile(p) and now - os.path.getmtime(p) < 1200:
                recent.append(f"{n}:{int(now-os.path.getmtime(p))}s")
    # also fixture
    fix = os.path.join(root, "corpus/fixtures/r99-window-lapse")
    fix_present = os.path.isdir(fix)
    line = f"=== try={i} {time.strftime('%Y-%m-%dT%H:%M:%S%z')} S5={s5 or 'NONE'} r99={r99 or 'NONE'} recent_knowledge={recent or 'NONE'} fixture={'Y' if fix_present else 'N'} seats={len(names)} ===\n"
    with open(log, "a") as f:
        f.write(line)
    print(line, end="")
    if len(r99) >= 2 and s5:
        print("READY")
        break
else:
    print("STILL_WAITING")
