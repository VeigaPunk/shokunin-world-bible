/bin/bash
-c
cd '/home/vgpnk/Projects/shokunin/lore-shokunin' && mkdir -p .ufo/waves/round-99/seats/V115-scratch && date '+%Y-%m-%d %H:%M:%S %z' > .ufo/waves/round-99/seats/V115-scratch/poll.log && echo "poll-start: S1 report absent; contract schema_version=1 pins=6 files=5; tool mtime newer than contract" >> .ufo/waves/round-99/seats/V115-scratch/poll.log && for i in $(seq 1 40); do
  date "+try $i %H:%M:%S" >> .ufo/waves/round-99/seats/V115-scratch/poll.log
  ls -1 .ufo/waves/round-99/seats/S1*.md >> .ufo/waves/round-99/seats/V115-scratch/poll.log 2>&1 || echo "S1-report: ABSENT" >> .ufo/waves/round-99/seats/V115-scratch/poll.log
  python3 - <<'PY' >> .ufo/waves/round-99/seats/V115-scratch/poll.log
import json
from pathlib import Path
c = json.loads(Path("corpus/tools/dependency-freeze-contract.json").read_text())
pins = c.get("pins", {})
sample = next(iter(pins.values())) if pins else {}
files_n = len(sample.get("files") or {})
print(f"contract schema={c.get('schema_version')} pins={sorted(pins)} files_n={files_n} payload_schema={sample.get('payload_schema_version')} has_r99={'r99-window-lapse' in pins}")
PY
  stat -c 'mtime py=%y contract=%y' corpus/tools/dependency_freeze.py corpus/tools/dependency-freeze-contract.json >> .ufo/waves/round-99/seats/V115-scratch/poll.log
  s1=$(ls .ufo/waves/round-99/seats/S1*.md 2>/dev/null | head -1)
  ready=$(python3 - <<'PY'
import json
from pathlib import Path
c = json.loads(Path("corpus/tools/dependency-freeze-contract.json").read_text())
pins = c.get("pins", {})
sample = next(iter(pins.values())) if pins else {}
ok = len(sample.get("files") or {}) >= 10 and sample.get("payload_schema_version") == 2
print("READY" if ok else "WAIT")
PY
)
  echo "ready=$ready s1=${s1:-ABSENT}" >> .ufo/waves/round-99/seats/V115-scratch/poll.log
  if [ -n "$s1" ] && [ "$ready" = "READY" ]; then
    echo "POLL-DONE both present" >> .ufo/waves/round-99/seats/V115-scratch/poll.log
    exit 0
  fi
  if [ "$i" -lt 40 ]; then
    sleep 20
  fi
done
echo "POLL-EXHAUSTED" >> .ufo/waves/round-99/seats/V115-scratch/poll.log
exit 3
