import os, time, hashlib, subprocess
from pathlib import Path
root = Path("/home/vgpnk/Projects/shokunin/lore-shokunin")
seats = root / ".ufo/waves/round-99/seats"
contract = root / "corpus/tools/dependency-freeze-contract.json"
tool = root / "corpus/tools/dependency_freeze.py"
s1_candidates = [
    seats / "S1-freeze-hygiene.md",
    seats / "S1.md",
    seats / "S1-c1.md",
]
base = subprocess.check_output(["git","show","4265df5:corpus/tools/dependency-freeze-contract.json"], cwd=root)
base_h = hashlib.sha256(base).hexdigest()
print(f"POLL_START {time.strftime('%Y-%m-%dT%H:%M:%S%z')} baseline_contract_sha {base_h}")
print(f"HEAD {subprocess.check_output(['git','rev-parse','HEAD'], cwd=root, text=True).strip()}")
for i in range(1, 41):
    now = time.strftime("%Y-%m-%dT%H:%M:%S%z")
    seat_exists = seats.is_dir()
    seat_files = []
    if seat_exists:
        try:
            seat_files = sorted(p.name for p in seats.iterdir() if p.is_file())
        except FileNotFoundError:
            seat_exists = False
            seat_files = []
    s1_hits = [str(p.relative_to(root)) for p in s1_candidates if p.is_file()]
    # also glob S1*.md if dir exists
    if seat_exists:
        s1_hits += [str(p.relative_to(root)) for p in seats.glob("S1*.md")]
        s1_hits = sorted(set(s1_hits))
    cur = contract.read_bytes() if contract.is_file() else b""
    cur_h = hashlib.sha256(cur).hexdigest() if cur else "MISSING"
    tool_h = hashlib.sha256(tool.read_bytes()).hexdigest() if tool.is_file() else "MISSING"
    changed = cur != base
    print(f"try={i} ts={now} seats_dir={seat_exists} n_files={len(seat_files)} s1={s1_hits or 'ABSENT'} contract_changed={changed} contract_sha={cur_h[:16]} tool_sha={tool_h[:16]}", flush=True)
    if s1_hits and changed:
        print("READY", flush=True)
        print("S1_REPORTS", *s1_hits, flush=True)
        print("SEAT_FILES", *seat_files, flush=True)
        break
    if i == 40:
        print("ABSENT_AFTER_40", flush=True)
        print("SEAT_FILES", *seat_files, flush=True)
        break
    time.sleep(20)
PY
