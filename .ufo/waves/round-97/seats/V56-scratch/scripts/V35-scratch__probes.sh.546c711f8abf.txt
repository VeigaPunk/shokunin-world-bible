#!/bin/bash
# V35 after-state probes — run on the FINAL revision (all producers landed).
# Never touches producer-owned files: all mutations happen on copies in scratch.
set -u
cd /home/vgpnk/Projects/shokunin/lore-shokunin
SC=.ufo/waves/round-97/seats/V35-scratch
P=$SC/probe
rm -rf "$P" && mkdir -p "$P/corpus/tools" "$P/corpus/fixtures"
cp corpus/tools/map_obligations.py corpus/tools/map-obligations-contract.json "$P/corpus/tools/"
cp corpus/tools/dependency_freeze.py corpus/tools/dependency-freeze-contract.json "$P/corpus/tools/"
cp -r corpus/fixtures/r95-crossing-aftermath corpus/fixtures/r96-crossing-response "$P/corpus/fixtures/"
mkdir -p "$P/foreign" "$P/tamper"
cp -r corpus/fixtures/r96-crossing-response "$P/foreign/r96-crossing-response"
cp -r corpus/fixtures/r95-crossing-aftermath "$P/foreign-r95"
MO="$P/corpus/tools/map_obligations.py"
MOC="$P/corpus/tools/map-obligations-contract.json"
DF="$P/corpus/tools/dependency_freeze.py"
DFC="$P/corpus/tools/dependency-freeze-contract.json"
O=$SC/probe-out; mkdir -p "$O"

echo "== C1a: contract drops dual_pointer pin (loader must now fail) =="
python3 - "$MOC" <<'EOF'
import json, sys
p = sys.argv[1]; c = json.load(open(p))
before = len(c["self_test"]["cases"])
c["self_test"]["cases"] = [x for x in c["self_test"]["cases"] if x["id"] != "dual_pointer"]
json.dump(c, open(p, "w"), indent=2); print(f"cases {before} -> {len(c['self_test']['cases'])}")
EOF
python3 "$MO" --self-test > "$O/c1a.txt" 2>&1; echo "c1a exit=$? (want 2)"; head -2 "$O/c1a.txt"
# restore
cp corpus/tools/map-obligations-contract.json "$MOC"

echo "== C1c: contract drops schema_drift (r96-added pin; must now fail too) =="
python3 - "$MOC" <<'EOF'
import json, sys
p = sys.argv[1]; c = json.load(open(p))
c["self_test"]["cases"] = [x for x in c["self_test"]["cases"] if x["id"] != "schema_drift"]
json.dump(c, open(p, "w"), indent=2); print("dropped schema_drift")
EOF
python3 "$MO" --self-test > "$O/c1c.txt" 2>&1; echo "c1c exit=$? (want 2)"; head -2 "$O/c1c.txt"
cp corpus/tools/map-obligations-contract.json "$MOC"

echo "== C1b: neutralize dual_pointer emission in tool copy (pin must be load-bearing) =="
sed 's/if mention is not None:/if False:/' "$MO" > "$MO.neutralized" && mv "$MO.neutralized" "$MO"
grep -c "if False:" "$MO"
python3 "$MO" --self-test > "$O/c1b.txt" 2>&1; echo "c1b exit=$? (want 2)"; grep -E "dual_pointer|RESULT" "$O/c1b.txt" | head -4
cp corpus/tools/map_obligations.py "$MO"   # restore original copy

echo "== C2a: foreign same-named byte-identical copy (want exit 2 PATH-MISMATCH) =="
python3 "$DF" "$P/foreign/r96-crossing-response" > "$O/c2a.txt" 2>&1; echo "c2a exit=$? (want 2)"; cat "$O/c2a.txt"

echo "== C2b: --freeze out-of-scope path (want exit 2, contract pins unchanged) =="
python3 -c "import json;print('pins before:',sorted(json.load(open('$DFC'))['pins']))" > "$O/c2b-pins-before.txt"
python3 "$DF" --freeze "$P/foreign-r95" > "$O/c2b.txt" 2>&1; echo "c2b exit=$? (want 2)"
python3 -c "import json;print('pins after: ',sorted(json.load(open('$DFC'))['pins']))" >> "$O/c2b-pins-before.txt"
cat "$O/c2b.txt" "$O/c2b-pins-before.txt"

echo "== C2c: digest tamper outside inventory (want DIGEST-MISMATCH naming timeline.json) =="
rm -rf "$P/tamper" && cp -r corpus/fixtures/r96-crossing-response "$P/tamper"
sed -i 's/Aru carries the north lantern/Aru carried the north lantern/' "$P/tamper/timeline.json"
python3 "$DF" "$P/tamper" > "$O/c2c.txt" 2>&1; echo "c2c exit=$? (want 2)"; cat "$O/c2c.txt"

echo "== C2d: pins record repo-relative paths =="
python3 - "$DFC" <<'EOF'
import json, sys
c = json.load(open(sys.argv[1]))
for name, pin in sorted(c["pins"].items()):
    print(name, "->", pin.get("fixture"), "revision", pin.get("revision","?")[:8])
EOF
echo done
