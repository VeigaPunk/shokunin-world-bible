#!/usr/bin/env bash
# V06 post-R3 battery. Writes ONLY under V06-scratch. Real contract write-protected by hash check.
set -u
LANE=/home/vgpnk/Projects/shokunin/lore-shokunin
SC=$LANE/.ufo/waves/round-97/seats/V06-scratch
cd "$LANE"

echo "### HASHES (delivered revision) ###"
sha256sum corpus/tools/dependency_freeze.py corpus/tools/dependency-freeze-contract.json

echo; echo "### B1: real self-test ###"
python3 corpus/tools/dependency_freeze.py --self-test; echo "EXIT=$?"

echo; echo "### B2: real --freeze on foreign scratch dir (guard must reject, contract must not change) ###"
BEFORE=$(sha256sum corpus/tools/dependency-freeze-contract.json | cut -d' ' -f1)
cp corpus/tools/dependency-freeze-contract.json "$SC/contract-backup-pre-probe.json"
rm -rf "$SC/foreign-target"; mkdir -p "$SC/foreign-target"
cp -r corpus/fixtures/r96-crossing-response/. "$SC/foreign-target/"
python3 corpus/tools/dependency_freeze.py --freeze "$SC/foreign-target"; echo "EXIT=$?"
AFTER=$(sha256sum corpus/tools/dependency-freeze-contract.json | cut -d' ' -f1)
echo "contract sha before=$BEFORE after=$AFTER"
[ "$BEFORE" = "$AFTER" ] && echo "NO-WRITE: OK" || echo "NO-WRITE: FAILED"

echo; echo "### B3: sandbox-r97 setup (tool+contract+fixture copies) ###"
SB=$SC/sandbox-r97
rm -rf "$SB"; mkdir -p "$SB/repo/corpus/tools" "$SB/repo/corpus/fixtures" "$SB/foreign"
cp corpus/tools/dependency_freeze.py "$SB/repo/corpus/tools/"
cp corpus/tools/dependency-freeze-contract.json "$SB/repo/corpus/tools/"
cp -r corpus/fixtures/r96-crossing-response "$SB/repo/corpus/fixtures/"
cp -r corpus/fixtures/r96-crossing-response "$SB/foreign/"
echo "sandbox ready"

echo; echo "### B3a: verify in-scope copy (expect 0) ###"
( cd "$SB/repo" && python3 corpus/tools/dependency_freeze.py corpus/fixtures/r96-crossing-response ); echo "EXIT=$?"

echo; echo "### B3b: verify foreign same-named copy (expect 2 PATH-MISMATCH) ###"
( cd "$SB/repo" && python3 corpus/tools/dependency_freeze.py "$SB/foreign/r96-crossing-response" ); echo "EXIT=$?"

echo; echo "### B3c: --freeze foreign dir in sandbox (expect 2, sandbox contract unchanged) ###"
SB_BEFORE=$(sha256sum "$SB/repo/corpus/tools/dependency-freeze-contract.json" | cut -d' ' -f1)
( cd "$SB/repo" && python3 corpus/tools/dependency_freeze.py --freeze "$SB/foreign/r96-crossing-response" ); echo "EXIT=$?"
SB_AFTER=$(sha256sum "$SB/repo/corpus/tools/dependency-freeze-contract.json" | cut -d' ' -f1)
echo "sandbox contract sha before=$SB_BEFORE after=$SB_AFTER"

echo; echo "### B3d: --freeze in-scope sandbox copy (expect 0, works) ###"
( cd "$SB/repo" && python3 corpus/tools/dependency_freeze.py --freeze corpus/fixtures/r96-crossing-response ); echo "EXIT=$?"
( cd "$SB/repo" && python3 corpus/tools/dependency_freeze.py corpus/fixtures/r96-crossing-response ); echo "EXIT=$?"

echo; echo "### B3e: digest tamper names the file (expect 2 DIGEST-MISMATCH naming timeline.json) ###"
python3 - "$SB/repo/corpus/fixtures/r96-crossing-response/timeline.json" <<'PYEOF'
import sys, pathlib
p = pathlib.Path(sys.argv[1])
t = p.read_text(encoding="utf-8")
p.write_text(t.replace("south bell at Salt Court", "South bell at Salt Court", 1), encoding="utf-8")
print("tampered:", p)
PYEOF
( cd "$SB/repo" && python3 corpus/tools/dependency_freeze.py corpus/fixtures/r96-crossing-response ); echo "EXIT=$?"

echo; echo "### B4: real chain verify (4 pinned fixtures, read-only) ###"
for f in r88-m3-maya-cluster r94-paired-crossing r95-crossing-aftermath r96-crossing-response; do
  echo "--- $f ---"
  python3 corpus/tools/dependency_freeze.py "corpus/fixtures/$f"; echo "EXIT=$?"
done

echo; echo "### B5: contract r96 pin inspection ###"
python3 - <<'PYEOF'
import json
c = json.load(open("corpus/tools/dependency-freeze-contract.json"))
pin = c["pins"]["r96-crossing-response"]
print("keys:", sorted(pin))
print("fixture:", pin.get("fixture"))
print("revision:", pin.get("revision"))
print("content_sha256:", pin.get("content_sha256"))
print("files:", json.dumps(pin.get("files"), indent=1, sort_keys=True))
print("self_test case count:", len(c.get("self_test", {}).get("expected", {})))
print("pins:", sorted(c["pins"]))
PYEOF
echo "battery done"
