#!/bin/bash
# V05 drift compare: baseline tool vs delivered tool, byte-identical stdout+exit expected.
set -u
R=/home/vgpnk/Projects/shokunin/lore-shokunin
S=$R/.ufo/waves/round-97/seats/V05-scratch
cd $R
BASE=$S/baseline/map_obligations.py
POST=$R/corpus/tools/map_obligations.py
FAIL=0
for f in "$S/dual-ref" "$S/dual-ref-colon" "corpus/fixtures/r87-myth-cluster" "corpus/fixtures/r88-m3-maya-cluster" "corpus/fixtures/r96-crossing-response"; do
  b=$(python3 "$BASE" "$f" 2>&1); be=$?
  p=$(python3 "$POST" "$f" 2>&1); pe=$?
  if [ "$b" = "$p" ] && [ "$be" = "$pe" ]; then
    echo "IDENTICAL exit=$be $f"
  else
    FAIL=1
    echo "DRIFT $f"
    echo "--- baseline exit=$be"; echo "$b"
    echo "--- post exit=$pe"; echo "$p"
  fi
done
exit $FAIL
