**R10 normalized-key scan.** Exact script run against the temporary candidate and each landed cluster:

```sh
python3 - "$cluster" <<'PY'
import json
import sys
import unicodedata
from pathlib import Path
candidate = Path(sys.argv[1])
root = Path('corpus/fixtures')
def key(value):
    value = unicodedata.normalize('NFKD', value).casefold()
    return ''.join(ch for ch in value if not unicodedata.combining(ch) and ch.isalnum())
def load(cluster, name):
    return json.loads((cluster / name).read_text(encoding='utf-8'))
def collect(cluster):
    entities = load(cluster, 'entities.json')
    sources = load(cluster, 'sources.json')
    timeline = load(cluster, 'timeline.json')
    relations = load(cluster, 'relations.json')
    ids = {key(row['id']) for row in entities}
    names = {key(row['name']) for row in entities}
    aliases = {key(alias) for row in entities for alias in row.get('aliases', [])}
    return {
        'ids': ids, 'names': names, 'aliases': aliases,
        'all': ids | names | aliases,
        'sources': {key(row['id']) for row in sources},
        'events': {key(row['id']) for row in timeline},
        'places': {key(row['place_id']) for row in timeline},
        'relations': {tuple(key(value) for value in (row['from'], row['type'], row['to'])) for row in relations},
    }
new = collect(candidate)
for existing_name in ('r87-myth-cluster', 'r88-m3-maya-cluster'):
    old = collect(root / existing_name)
    print(f'{existing_name}: entity_id_intersections={sorted(new["ids"] & old["ids"])}; display_name_intersections={sorted(new["names"] & old["names"])}; alias_intersections={sorted(new["aliases"] & old["aliases"])}; all_entity_key_intersections={sorted(new["all"] & old["all"])}; source_id_intersections={sorted(new["sources"] & old["sources"])} (namespace only); timeline_event_id_intersections={sorted(new["events"] & old["events"])}; place_id_intersections={sorted(new["places"] & old["places"])}; relation_tuple_intersections={sorted(new["relations"] & old["relations"])}')
PY
```
