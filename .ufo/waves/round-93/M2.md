```sh

import json, os, unicodedata
from pathlib import Path

def key(value):
    value = unicodedata.normalize("NFKD", value).casefold()
    return "".join(ch for ch in value if not unicodedata.combining(ch) and ch.isalnum())

def tokens(value):
    value = unicodedata.normalize("NFKD", value).casefold()
    return "".join(ch if ch.isalnum() else " " for ch in value if not unicodedata.combining(ch)).split()

def load(root):
    read = lambda name: json.loads((root / name).read_text(encoding="utf-8"))
    entities, relations, timeline, sources = (read(name) for name in ("entities.json", "relations.json", "timeline.json", "sources.json"))
    return {
        "ids": {key(e["id"]) for e in entities},
        "labels": {key(value) for e in entities for value in [e["name"], *e.get("aliases", [])]},
        "events": {key(e["id"]) for e in timeline},
        "endpoints": ({key(r[field]) for r in relations for field in ("from", "to")} | {key(e[field]) for e in timeline for field in ("entity_id", "place_id")} | {key(dep) for e in timeline for dep in e["depends_on"]}),
        "sources": {(root.name, key(s["id"])) for s in sources},
        "entities": entities,
    }

root = Path(os.environ["M2_ROOT"])
m2 = load(root)
print("M2 normalized IDs=" + ",".join(sorted(m2["ids"])))
print("M2 normalized display-name/alias keys=" + ",".join(sorted(m2["labels"])))
print("M2 normalized event IDs=" + ",".join(sorted(m2["events"])))
print("M2 normalized relation/timeline endpoints=" + ",".join(sorted(m2["endpoints"])))
for name in ("r87-myth-cluster", "r88-m3-maya-cluster"):
    other_root = Path("corpus/fixtures") / name
    other = load(other_root)
    cross = (m2["ids"] & other["labels"]) | (m2["labels"] & other["ids"])
    print(f"{name}: ID-ID={sorted(m2['ids'] & other['ids'])}; label-label={sorted(m2['labels'] & other['labels'])}; ID-label={sorted(cross)}; event-ID={sorted(m2['events'] & other['events'])}; endpoint-to-entity-key={sorted(m2['endpoints'] & (other['ids'] | other['labels']))}; path-qualified-source-ID={sorted(m2['sources'] & other['sources'])}")
    docs = [(p, tokens(p.read_text(encoding="utf-8"))) for p in root.rglob("*") if p.is_file() and p.suffix in {".md", ".json"}]

            if hits:
                mentions.append((label, hits))
    print(f"{name}: prior display-name/alias text mentions={mentions}")
```
