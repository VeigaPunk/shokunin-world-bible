cd '/home/vgpnk/Projects/shokunin' && mkdir -p /home/vgpnk/Projects/shokunin/lore-shokunin/.ufo/waves/round-99/seats/V54-scratch && date -u +%Y-%m-%dT%H:%M:%SZ | tee /home/vgpnk/Projects/shokunin/lore-shokunin/.ufo/waves/round-99/seats/V54-scratch/round-open-utc.txt && date +%Y-%m-%dT%H:%M:%S%z | tee /home/vgpnk/Projects/shokunin/lore-shokunin/.ufo/waves/round-99/seats/V54-scratch/monitor-start-local.txt && python3 - << 'PY'
import os, time, glob
seats = "/home/vgpnk/Projects/shokunin/lore-shokunin/.ufo/waves/round-99/seats"
log = os.path.join(seats, "V54-scratch/curve.log")
seen = set()
start = time.strftime("%Y-%m-%dT%H:%M:%S%z")
with open(log, "a") as f:
    f.write(f"# sampler start {start} cadence=20s cap=900s seat=V54\n")
    f.flush()
deadline = time.time() + 900
while time.time() < deadline:
    now = time.strftime("%Y-%m-%dT%H:%M:%S%z")
    files = sorted(os.path.basename(p) for p in glob.glob(os.path.join(seats, "*.md")))
    new = [n for n in files if n not in seen]
    seen.update(files)
    line = f"{now} count={len(files)} new={','.join(new) if new else '-'}\n"
    with open(log, "a") as f:
        f.write(line)
        f.flush()
    print(line, end="", flush=True)
    if len(files) >= 127:  # subtract SELF: V54 report not yet written
        with open(log, "a") as f:
            f.write(f"# NONSELF_COMPLETE {now} count={len(files)} (SELF excluded by construction)\n")
        break
    time.sleep(20)
else:
    with open(log, "a") as f:
        f.write(f"# SAMPLER_CAP {time.strftime('%Y-%m-%dT%H:%M:%S%z')} count={len(seen)}\n")
PY
