**C9 source-content check.** The exact AST probe used for the pinned checker:

```sh
python3 - <<'PY'
import ast
from pathlib import Path
p = Path('corpus/fixtures/check_bible.py')
s = p.read_text()
tree = ast.parse(s)
tick = []
null = []
for node in ast.walk(tree):
    if isinstance(node, ast.If) and isinstance(node.test, ast.Compare):
        left = node.test.left
        comparators = node.test.comparators
        ops = node.test.ops
        if (isinstance(left, ast.Name) and left.id == 'tick'
                and len(comparators) == 1 and isinstance(comparators[0], ast.Name)
                and comparators[0].id == 'previous_tick'
                and len(ops) == 1 and isinstance(ops[0], ast.LtE)):
            source = ast.get_source_segment(s, node)
            tick.append((node.lineno, node.end_lineno, 'holder_identity_detail' not in source,
                         'chain_has_ambiguous_source' in source))
    if isinstance(node, ast.FunctionDef) and node.name == 'holder_identity_detail':
        for child in ast.walk(node):
            if isinstance(child, ast.If) and isinstance(child.test, ast.Compare):
                if (isinstance(child.test.left, ast.Name) and child.test.left.id == 'holder_id'
                        and len(child.test.comparators) == 1 and isinstance(child.test.comparators[0], ast.Constant)
                        and child.test.comparators[0].value is None):
                    null.append((child.lineno, child.end_lineno, any(isinstance(x, ast.Return) and isinstance(x.value, ast.Constant) and x.value.value is None for x in child.body)))
print(f'tick_comparison_nodes={tick!r}')
print(f'null_guard_nodes={null!r}')
assert len(tick) == 1 and tick[0][2] and tick[0][3]
assert len(null) == 1 and null[0][2]
print('c9_checker_ast=PASS')
```
