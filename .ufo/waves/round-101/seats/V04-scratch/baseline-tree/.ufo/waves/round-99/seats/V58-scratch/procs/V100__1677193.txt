import os, time, glob
from pathlib import Path
root = Path("/home/vgpnk/Projects/shokunin/lore-shokunin")
seats = root / ".ufo/waves/round-99/seats"
scratch = seats / "V100-scratch"
scratch.mkdir(parents=True, exist_ok=True)
log = scratch / "poll.log"
arts = [
    root / "corpus/tools/dependency_freeze.py",
    root / "corpus/tools/dependency-freeze-contract.json",
    root / "corpus/tools/README.md",
    root / "SKILL.md",
    root / "corpus/fixtures/r99-window-lapse",
    root / "corpus/knowledge/r99-gate-blind-spot-table.md",
    root / "corpus/knowledge/r99-lapse-scope-note.md",
]
fixture_files = [
    "entities.json","relations.json","timeline.json","sources.json","contradictions.json",
    "README.md","claim-excerpt-map.md","fictional-excerpts.md","inspiration-note.md","source-excerpt.md",
]
# expected producer report prefixes
producers = ["S1","S2","S3","S4","S5"]
with open(log, "a") as f:
    f.write(f"POLL_START {time.strftime('%Y-%m-%dT%H:%M:%SZ', time.gmtime())}\n")
for i in range(1, 41):
    ts = time.strftime("%Y-%m-%dT%H:%M:%SZ", time.gmtime())
    reports = sorted(p.name for p in seats.glob("*.md")) if seats.is_dir() else []
    # producer reports: S*.md at seats root
    prod = [n for n in reports if n.startswith("S")]
    missing_prod = []
    for s in producers:
        if not any(n.startswith(s) for n in prod):
            missing_prod.append(s)
    miss_art = []
    for a in arts:
        if not a.exists():
            miss_art.append(str(a.relative_to(root)))
    fix = root / "corpus/fixtures/r99-window-lapse"
    miss_fix = []
    if fix.is_dir():
        for fn in fixture_files:
            if not (fix / fn).is_file():
                miss_fix.append(fn)
    else:
        miss_fix = ["DIR"]
    line = (f"try={i} t={ts} reports={len(reports)} prod={prod or 'NONE'} "
            f"missProd={missing_prod or '[]'} missArt={miss_art or '[]'} missFix={miss_fix or '[]'}")
    with open(log, "a") as f:
        f.write(line + "\n")
        f.write("  present: " + " ".join(reports) + "\n")
    print(line, flush=True)
    # keep polling the full window so the log captures the landing curve;
    # do not exit early — bookkeeper needs the curve through cutoff.
    if i < 40:
        time.sleep(20)
with open(log, "a") as f:
    f.write(f"POLL_END {time.strftime('%Y-%m-%dT%H:%M:%SZ', time.gmtime())} tries=40\n")
print("POLL_DONE", flush=True)
PY
