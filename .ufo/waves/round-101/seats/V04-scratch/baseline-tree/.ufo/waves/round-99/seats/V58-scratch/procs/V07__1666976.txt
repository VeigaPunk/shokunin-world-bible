import os, time, hashlib
from pathlib import Path
root = Path("/home/vgpnk/Projects/shokunin/lore-shokunin")
seats = root / ".ufo/waves/round-99/seats"
log = seats / "V07-scratch/poll.log"
log.parent.mkdir(parents=True, exist_ok=True)
targets = [
    root / "corpus/tools/dependency_freeze.py",
    root / "corpus/tools/dependency-freeze-contract.json",
]
baseline = {
    "dependency_freeze.py": "03d04365cf00903e783562d8d760a0f751e477af218b79e6484ab34a9315a24b",
    "dependency-freeze-contract.json": "153ebc5b2092948ef43a8693bc1ebd6a0ace80d798210a8d54c96d6e0573a947",
}
def sha(p):
    h = hashlib.sha256()
    with open(p, "rb") as f:
        for chunk in iter(lambda: f.read(1<<20), b""):
            h.update(chunk)
    return h.hexdigest()
for i in range(1, 41):
    now = time.strftime("%Y-%m-%dT%H:%M:%S%z")
    seat_files = []
    if seats.exists():
        seat_files = sorted(p.name for p in seats.iterdir() if p.is_file())
    s1 = [n for n in seat_files if n.startswith("S1")]
    hashes = {}
    changed = []
    for t in targets:
        if t.exists():
            h = sha(t)
            hashes[t.name] = h
            if h != baseline.get(t.name):
                changed.append(t.name)
    line = f"try={i} ts={now} s1={s1 or 'NONE'} seats_n={len(seat_files)} changed={changed or 'NONE'} freeze={hashes.get('dependency_freeze.py','MISSING')[:12]} contract={hashes.get('dependency-freeze-contract.json','MISSING')[:12]}"
    with open(log, "a") as f:
        f.write(line + "\n")
        if seat_files:
            f.write("  files: " + ", ".join(seat_files[:40]) + "\n")
    print(line, flush=True)
    # Producer report AND code change both required before review starts.
    if s1 and changed:
        print("READY", flush=True)
        break
    if i < 40:
        time.sleep(20)
else:
    print("ABSENT_AFTER_40", flush=True)
