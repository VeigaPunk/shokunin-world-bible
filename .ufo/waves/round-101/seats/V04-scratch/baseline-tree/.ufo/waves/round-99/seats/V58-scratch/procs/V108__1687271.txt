/bin/bash
-c
cd '/home/vgpnk/Projects/shokunin' && set -u
ROOT=/home/vgpnk/Projects/shokunin/lore-shokunin
SEATS="$ROOT/.ufo/waves/round-99/seats"
SCRATCH="$SEATS/V108-scratch"
mkdir -p "$SCRATCH"
LOG="$SCRATCH/poll.log"
echo "POLL-START $(date -Is)" | tee "$LOG"
for i in $(seq 1 40); do
  echo "===== TRY $i $(date -Is) =====" | tee -a "$LOG"
  python3 - <<'PY' | tee -a "$LOG"
import os, hashlib
from pathlib import Path
root = Path("/home/vgpnk/Projects/shokunin/lore-shokunin")
seats = root / ".ufo/waves/round-99/seats"
tool = root / "corpus/tools/dependency_freeze.py"
contract = root / "corpus/tools/dependency-freeze-contract.json"
fix = root / "corpus/fixtures/r99-window-lapse"
need_reports = [
    "S1-freeze-hygiene.md",
    "S2-tools-docs.md",
    "S3-fixture-author.md",
    "S4-fixture-docs.md",
    "S5-knowledge-notes.md",
]
print("tool_exists", tool.is_file(), "tool_mtime", tool.stat().st_mtime if tool.is_file() else None)
print("tool_sha", hashlib.sha256(tool.read_bytes()).hexdigest() if tool.is_file() else None)
print("tool_bytes", tool.stat().st_size if tool.is_file() else None)
text = tool.read_text(encoding="utf-8") if tool.is_file() else ""
markers = ["markdown_tamper", "PINNED", "prose", "schema_version", "source-excerpt", "claim-excerpt-map", "inspiration-note"]
for m in markers:
    print(f"marker {m}: {m in text}")
print("FROZEN_CASES_count_guess", text.count('"') )
# count tuple members of FROZEN_CASES roughly
if "FROZEN_CASES" in text:
    start = text.find("FROZEN_CASES")
    snippet = text[start:start+1200]
    print("FROZEN_SNIPPET_HEAD")
    print(snippet[:800])
print("contract_exists", contract.is_file())
if contract.is_file():
    c = contract.read_text(encoding="utf-8")
    print("contract_sha", hashlib.sha256(contract.read_bytes()).hexdigest(), "bytes", contract.stat().st_size)
    print("schema_version_line", [ln.strip() for ln in c.splitlines() if "schema_version" in ln][:5])
    print("has_r99_pin", "r99-window-lapse" in c)
    print("has_readme_key", '"README.md"' in c or "'README.md'" in c)
print("fixture_dir", fix.is_dir())
if fix.is_dir():
    files = sorted(p.name for p in fix.iterdir() if p.is_file())
    print("fixture_files", files)
for name in need_reports:
    p = seats / name
    print(f"report {name}: exists={p.is_file()} size={p.stat().st_size if p.is_file() else 0}")
# also list any S*.md
smds = sorted(p.name for p in seats.glob("S*.md"))
print("S_md", smds)
PY
  ready=$(python3 - <<'PY'
from pathlib import Path
root = Path("/home/vgpnk/Projects/shokunin/lore-shokunin")
tool = (root / "corpus/tools/dependency_freeze.py").read_text(encoding="utf-8")
contract = (root / "corpus/tools/dependency-freeze-contract.json").read_text(encoding="utf-8")
seats = root / ".ufo/waves/round-99/seats"
reports = all((seats / n).is_file() and (seats / n).stat().st_size > 0 for n in [
    "S1-freeze-hygiene.md","S2-tools-docs.md","S3-fixture-author.md","S4-fixture-docs.md","S5-knowledge-notes.md"])
# C1 signal: ten-file pin language or markdown tamper case present in tool OR contract pins README
c1 = ("markdown_tamper" in tool) or ('"README.md"' in contract and "r99-window-lapse" in contract and "source-excerpt.md" in tool)
print("YES" if (reports and c1) else "NO")
PY
)
  echo "READY=$ready" | tee -a "$LOG"
  if [ "$ready" = "YES" ]; then
    echo "POLL-READY try=$i $(date -Is)" | tee -a "$LOG"
    exit 0
  fi
  if [ "$i" -lt 40 ]; then
    sleep 20
  fi
done
echo "POLL-EXHAUSTED $(date -Is)" | tee -a "$LOG"
exit 3
