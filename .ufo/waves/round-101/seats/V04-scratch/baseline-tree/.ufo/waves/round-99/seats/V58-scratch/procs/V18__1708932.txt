/bin/bash
-c
cd '/home/vgpnk/Projects/shokunin' && # poll loop: 20s cadence, max 40 tries; exit 0 when table AND S5 report exist and table is non-stub
set -u
SCRATCH=/home/vgpnk/Projects/shokunin/lore-shokunin/.ufo/waves/round-99/seats/V18-scratch
LOG=$SCRATCH/poll.log
TABLE=/home/vgpnk/Projects/shokunin/lore-shokunin/corpus/knowledge/r99-gate-blind-spot-table.md
SEATS=/home/vgpnk/Projects/shokunin/lore-shokunin/.ufo/waves/round-99/seats
echo "POLL_START $(date -Iseconds)" >> "$LOG"
for i in $(seq 1 40); do
  reports=$(ls -1 "$SEATS"/S5*.md 2>/dev/null | grep -v scratch || true)
  tsize=0
  texists=no
  if [ -f "$TABLE" ]; then
    texists=yes
    tsize=$(wc -c < "$TABLE")
  fi
  # non-stub: >800 bytes and contains a markdown table row or the word blind
  substantive=no
  if [ "$texists" = yes ] && [ "$tsize" -gt 800 ]; then
    if grep -qE 'depends_on|blind|DIGEST|regenerat' "$TABLE"; then
      substantive=yes
    fi
  fi
  echo "TRY $i $(date -Iseconds) table=$texists size=$tsize substantive=$substantive reports=${reports:-NONE}" >> "$LOG"
  if [ "$substantive" = yes ] && [ -n "$reports" ]; then
    echo "POLL_HIT try=$i $(date -Iseconds)" >> "$LOG"
    echo "HIT try=$i"
    exit 0
  fi
  sleep 20
done
echo "POLL_EXHAUSTED $(date -Iseconds)" >> "$LOG"
echo EXHAUSTED
exit 2
