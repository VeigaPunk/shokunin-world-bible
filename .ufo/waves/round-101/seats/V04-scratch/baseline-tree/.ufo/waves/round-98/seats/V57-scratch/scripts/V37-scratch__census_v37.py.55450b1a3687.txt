#!/usr/bin/env python3
"""V37 (round 98) independent census replica.

Runs corpus/fixtures/check_bible.py repo-relative over every fixture directory
under a given root (working tree or extracted git archive) and emits:
  - summary: dirs / clean(exit0) / negative(exit1) / violation lines under two
    counting conventions / non-0-1 exits
  - per-dir receipt: exit code + sha256 of combined stdout+stderr, for tuple
    comparison between the 9a3d648 archive and the delivered tree.

Independent implementation (not a copy of any other seat's census script).
"""
import hashlib
import json
import os
import subprocess
import sys


def census(root, receipts_path):
    fixroot = os.path.join(root, "corpus", "fixtures")
    dirs = sorted(
        d for d in os.listdir(fixroot)
        if os.path.isdir(os.path.join(fixroot, d)) and not d.startswith(".")
    )
    receipts = {}
    clean = neg = 0
    v_dash = v_word = 0
    odd = []
    env = dict(os.environ, PYTHONDONTWRITEBYTECODE="1")
    for d in dirs:
        rel = "corpus/fixtures/" + d
        p = subprocess.run(
            [sys.executable, "corpus/fixtures/check_bible.py", rel],
            cwd=root, capture_output=True, text=True, env=env,
        )
        out = p.stdout + p.stderr
        n_dash = sum(1 for line in out.splitlines() if line.startswith("- "))
        n_word = sum(1 for line in out.splitlines() if "VIOLATION" in line)
        receipts[rel] = {
            "exit": p.returncode,
            "sha256": hashlib.sha256(out.encode()).hexdigest(),
            "dash_lines": n_dash,
            "violation_word_lines": n_word,
        }
        if p.returncode == 0:
            clean += 1
        elif p.returncode == 1:
            neg += 1
        else:
            odd.append([rel, p.returncode, out.strip()[:300]])
        v_dash += n_dash
        v_word += n_word
    summary = {
        "root": root,
        "fixture_dirs": len(dirs),
        "clean_exit0": clean,
        "negative_exit1": neg,
        "violation_lines_dash": v_dash,
        "violation_lines_word": v_word,
        "non01": odd,
    }
    with open(receipts_path, "w") as fh:
        json.dump(receipts, fh, indent=1, sort_keys=True)
    return summary


if __name__ == "__main__":
    root = sys.argv[1]
    receipts_path = sys.argv[2]
    print(json.dumps(census(os.path.abspath(root), receipts_path)))
