/bin/bash
-c
cd '/home/vgpnk/Projects/shokunin' && set -u
LANE=/home/vgpnk/Projects/shokunin/lore-shokunin
SEATS="$LANE/.ufo/waves/round-99/seats"
SCRATCH="$SEATS/V106-scratch"
mkdir -p "$SCRATCH"
LOG="$SCRATCH/poll.log"
: > "$LOG"
for i in $(seq 1 40); do
  ts=$(date -Iseconds)
  py_mtime=$(stat -c '%y' "$LANE/corpus/tools/dependency_freeze.py" 2>/dev/null || echo missing)
  ct_mtime=$(stat -c '%y' "$LANE/corpus/tools/dependency-freeze-contract.json" 2>/dev/null || echo missing)
  s1=$(ls "$SEATS"/S1*.md 2>/dev/null | tr '\n' ' ')
  schema=$(python3 -c '
import json
p="/home/vgpnk/Projects/shokunin/lore-shokunin/corpus/tools/dependency-freeze-contract.json"
c=json.load(open(p))
pins=c.get("pins") or {}
nf=sorted({len((p.get("files") or {})) for p in pins.values()}) if pins else []
cases=sorted((c.get("self_test") or {}).get("expected") or {})
print("schema", c.get("schema_version"), "pins", len(pins), "nfiles", nf, "ncases", len(cases), "md_tamper", "markdown_tamper" in cases or "md_tamper" in cases or any("markdown" in x or "prose" in x for x in cases))
' 2>&1)
  line="$ts try=$i s1=[${s1:-ABSENT}] py=$py_mtime ct=$ct_mtime $schema"
  echo "$line" | tee -a "$LOG"
  # delivery signal: contract schema bumped OR every pin has 10 files OR S1 report exists AND pins have 10 files
  ready=$(python3 -c '
import json,sys
p="/home/vgpnk/Projects/shokunin/lore-shokunin/corpus/tools/dependency-freeze-contract.json"
c=json.load(open(p))
pins=c.get("pins") or {}
if not pins:
    sys.exit(1)
nfs=[len((pin.get("files") or {})) for pin in pins.values()]
if min(nfs) >= 10 and c.get("schema_version", 1) != 1:
    sys.exit(0)
if min(nfs) >= 10:
    # ten-file payload is the C1 delivery even if schema field name differs; require schema change OR a version key
    if c.get("schema_version") not in (None, 1) or c.get("pin_schema") or c.get("payload_schema"):
        sys.exit(0)
    # ten files is itself the claim; accept if all pins have 10 file digests
    sys.exit(0)
sys.exit(1)
' && echo READY || echo WAIT)
  if [ "$ready" = "READY" ]; then
    echo "READY at try $i" | tee -a "$LOG"
    exit 0
  fi
  # also accept S1 report + ten-file pins
  if [ -n "${s1:-}" ]; then
    ten=$(python3 -c '
import json,sys
c=json.load(open("/home/vgpnk/Projects/shokunin/lore-shokunin/corpus/tools/dependency-freeze-contract.json"))
pins=c.get("pins") or {}
nfs=[len((pin.get("files") or {})) for pin in pins.values()]
sys.exit(0 if pins and min(nfs)>=10 else 1)
' && echo yes || echo no)
    if [ "$ten" = "yes" ]; then
      echo "READY via S1 report + 10-file pins at try $i" | tee -a "$LOG"
      exit 0
    fi
  fi
  sleep 20
done
echo "ABSENT after 40 tries" | tee -a "$LOG"
exit 3
