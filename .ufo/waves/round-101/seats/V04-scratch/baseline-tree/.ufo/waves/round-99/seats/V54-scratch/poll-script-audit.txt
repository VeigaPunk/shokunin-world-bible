#!/bin/sh
# V08 c1-negative-rerun poller. 20s cadence, 40 tries.
# Hit requires: S1 report present AND delivered freeze tool+contract differ from
# r98 baseline shas AND combined sha stable for 3 consecutive checks.
set -u
LANE=/home/vgpnk/Projects/shokunin/lore-shokunin
SEATS="$LANE/.ufo/waves/round-99/seats"
TOOL="$LANE/corpus/tools/dependency_freeze.py"
CONTRACT="$LANE/corpus/tools/dependency-freeze-contract.json"
BASE_TOOL=03d04365cf00903e783562d8d760a0f751e477af218b79e6484ab34a9315a24b
BASE_CONTRACT=153ebc5b2092948ef43a8693bc1ebd6a0ace80d798210a8d54c96d6e0573a947
LOG="$LANE/.ufo/waves/round-99/seats/V08-scratch/poll.log"
: > "$LOG"
prev=""
stable=0
i=1
while [ "$i" -le 40 ]; do
  ts=$(date '+%H:%M:%S')
  report=$(ls -1 "$SEATS"/S1-*.md 2>/dev/null | head -1)
  report=${report:-none}
  if [ -f "$TOOL" ]; then
    toolsha=$(sha256sum "$TOOL" | awk '{print $1}')
  else
    toolsha=missing
  fi
  if [ -f "$CONTRACT" ]; then
    ctsha=$(sha256sum "$CONTRACT" | awk '{print $1}')
  else
    ctsha=missing
  fi
  if [ "$toolsha" = "$BASE_TOOL" ]; then toolstate=same; else toolstate=changed; fi
  if [ "$ctsha" = "$BASE_CONTRACT" ]; then ctstate=same; else ctstate=changed; fi
  echo "try=$i t=$ts report=$report tool=$toolstate contract=$ctstate toolsha=${toolsha} ctsha=${ctsha}" | tee -a "$LOG"
  if [ "$report" != none ] && [ "$toolstate" = changed ] && [ "$ctstate" = changed ] && [ "$toolsha" != missing ] && [ "$ctsha" != missing ]; then
    combo=$(printf '%s\n%s\n' "$toolsha" "$ctsha" | sha256sum | awk '{print $1}')
    if [ "$combo" = "$prev" ]; then
      stable=$((stable+1))
    else
      stable=0
      prev=$combo
    fi
    echo "STAB i=$i sha=$combo stable=$stable" | tee -a "$LOG"
    if [ "$stable" -ge 2 ]; then
      echo "POLL_HIT try=$i report=$report tool=$toolsha contract=$ctsha" | tee -a "$LOG"
      echo "POLL_STABLE sha=$combo" | tee -a "$LOG"
      exit 0
    fi
  else
    prev=""
    stable=0
  fi
  i=$((i+1))
  if [ "$i" -le 40 ]; then
    sleep 20
  fi
done
echo "POLL_ABSENT tries=40" | tee -a "$LOG"
exit 2

#!/bin/bash

: > "$LOG"
hash_of() { sha256sum "$1" 2>/dev/null | awk '{print $1}'; }

r98_readme=$(hash_of "$R98/README.md")
echo "poll-start $(date '+%H:%M:%S') r98_map=${r98_map:0:16} r98_readme=${r98_readme:0:16}" | tee -a "$LOG"
for try in $(seq 1 40); do
  ts=$(date '+%H:%M:%S')
  map_state=absent
  readme_state=absent
  if [ -f "$FIX/claim-excerpt-map.md" ]; then
    h=$(hash_of "$FIX/claim-excerpt-map.md")
    if [ "$h" = "$r98_map" ]; then map_state=same-as-r98; else map_state=moved; fi
  fi
  if [ -f "$FIX/README.md" ]; then
    h=$(hash_of "$FIX/README.md")
    if [ "$h" = "$r98_readme" ]; then readme_state=same-as-r98; else readme_state=moved; fi
  fi
  reports=""
  for f in "$SEATS"/S4*.md; do
    [ -f "$f" ] || continue
    base=$(basename "$f")
    case "$base" in
      V12*) continue ;;
    esac
    reports="$reports $base"
  done
  reports=${reports# }
  [ -z "$reports" ] && reports=none
  line="try=$try t=$ts map=$map_state readme=$readme_state reports=$reports"
  echo "$line" | tee -a "$LOG"
  if [ "$map_state" = moved ] && [ "$readme_state" = moved ] && [ "$reports" != none ]; then
    echo "READY try=$try t=$ts" | tee -a "$LOG"
    exit 0
  fi
  sleep 20
done
echo "ABSENT after 40 tries t=$(date '+%H:%M:%S')" | tee -a "$LOG"
