import time, os, re
from pathlib import Path
from datetime import datetime

seats = Path("/home/vgpnk/Projects/shokunin/lore-shokunin/.ufo/waves/round-99/seats")
log = seats / "V101-scratch" / "poll.log"
lane = Path("/home/vgpnk/Projects/shokunin/lore-shokunin")
roster = [f"S{i}" for i in range(1,6)] + [f"V{i:02d}" if i < 100 else f"V{i}" for i in range(1,124)]
roster = [s for s in roster if s != "V101"]
assert len(roster) == 127, len(roster)
pref_re = re.compile(r"^(S[1-5]|V\d+)-.+\.md$")
artifacts = [
    "corpus/tools/dependency_freeze.py",
    "corpus/tools/dependency-freeze-contract.json",
    "corpus/tools/README.md",
    "SKILL.md",
    "corpus/fixtures/r99-window-lapse/README.md",
    "corpus/fixtures/r99-window-lapse/claim-excerpt-map.md",
    "corpus/fixtures/r99-window-lapse/contradictions.json",
    "corpus/fixtures/r99-window-lapse/entities.json",
    "corpus/fixtures/r99-window-lapse/fictional-excerpts.md",
    "corpus/fixtures/r99-window-lapse/inspiration-note.md",
    "corpus/fixtures/r99-window-lapse/relations.json",
    "corpus/fixtures/r99-window-lapse/source-excerpt.md",
    "corpus/fixtures/r99-window-lapse/sources.json",
    "corpus/fixtures/r99-window-lapse/timeline.json",
    "corpus/knowledge/r99-gate-blind-spot-table.md",
    "corpus/knowledge/r99-lapse-scope-note.md",
]
producers = ["S1","S2","S3","S4","S5"]
# 30 min / 20s = 90 tries. Record stall, do not treat as fatal.
MAX = 90
prev = -1
stable = 0
with log.open("a") as fh:
    fh.write(f"START {datetime.now().astimezone().isoformat()} roster_others=127 max_tries={MAX}\n")
    fh.flush()
    for n in range(1, MAX+1):
        mds = [p.name for p in seats.glob("*.md") if pref_re.match(p.name) and not p.name.startswith("V101-")]
        prefs = {}
        for name in mds:
            pref = name.split("-",1)[0]
            prefs.setdefault(pref, []).append(name)
        present = sorted(prefs)
        missing = [s for s in roster if s not in prefs]
        extra = [p for p in present if p not in roster and p != "V101"]
        dups = {k:v for k,v in prefs.items() if len(v)>1}
        art_missing = [a for a in artifacts if not (lane/a).exists()]
        prod_missing = [s for s in producers if s not in prefs]
        receipt = (lane/".ufo/waves/round-99/receipt.json").exists()
        line = (f"try={n} ts={datetime.now().astimezone().strftime('%H:%M:%S')} "
                f"others={len(present)} missing={len(missing)} dups={len(dups)} "
                f"extra={extra} prod_missing={prod_missing} art_missing={len(art_missing)} "
                f"receipt={int(receipt)}\n")
        fh.write(line)
        if missing and len(missing) <= 20:
            fh.write("  missing_seats=" + ",".join(missing) + "\n")
        if art_missing:
            fh.write("  art_missing=" + ",".join(art_missing) + "\n")
        fh.flush()
        if len(present) == prev:
            stable += 1
        else:
            stable = 0
            prev = len(present)
        # ready when 127 others present
        if len(missing) == 0 and len(dups) == 0:
            fh.write(f"READY others=127 at try={n}\n")
            fh.flush()
            break
        # producer-artifact stall note at try 40, but keep polling reports
        if n == 40 and (prod_missing or art_missing):
            fh.write(f"ARTIFACT_STALL_NOTE try=40 prod_missing={prod_missing} art_missing={art_missing} — continuing report poll\n")
            fh.flush()
        if n == MAX:
            fh.write(f"STALL_ESCAPE try={n} others={len(present)} missing={len(missing)} stable_tries={stable} — exiting poll, audit will record ABSENT\n")
            fh.write("MISSING=" + ",".join(missing) + "\n")
            fh.flush()
            break
        time.sleep(20)
print("poll-done")
