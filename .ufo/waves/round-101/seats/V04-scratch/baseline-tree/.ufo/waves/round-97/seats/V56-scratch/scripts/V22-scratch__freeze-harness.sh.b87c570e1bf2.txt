#!/usr/bin/env bash
# V22 freeze-harness: exercise the DELIVERED dependency_freeze.py against
# scratch copies of the r97 fixture, with the delivered path binding satisfied.
#
# Why a harness: the delivered pin binds the canonical corpus path, and
# verifier seats may not write under corpus/. So we copy the delivered tool +
# contract byte-for-byte into a scratch ROOT whose own corpus/fixtures/ holds
# the r97 copies; --freeze then records harness-relative paths and verify
# compares like with like.
set -u
REPO=/home/vgpnk/Projects/shokunin/lore-shokunin
S=$REPO/.ufo/waves/round-97/seats/V22-scratch
H=$S/harness
TOOL=$REPO/corpus/tools/dependency_freeze.py
CONTRACT=$REPO/corpus/tools/dependency-freeze-contract.json
FIX=$REPO/corpus/fixtures/r97-crossing-acknowledgment

echo "=== harness setup ==="
rm -rf "$H"
mkdir -p "$H/corpus/tools" "$H/corpus/fixtures"
cp "$TOOL" "$H/corpus/tools/dependency_freeze.py"
cp "$CONTRACT" "$H/corpus/tools/dependency-freeze-contract.json"
echo "tool copy sha: $(sha256sum "$TOOL" | cut -d' ' -f1)"
echo "tool copy sha (harness): $(sha256sum "$H/corpus/tools/dependency_freeze.py" | cut -d' ' -f1)"

# clean copy: pinned through the guarded --freeze path
cp -r "$FIX" "$H/corpus/fixtures/r97-clean"
echo
echo "=== freeze clean copy (guarded path) ==="
echo "\$ python3 \$H/corpus/tools/dependency_freeze.py --freeze \$H/corpus/fixtures/r97-clean"
python3 "$H/corpus/tools/dependency_freeze.py" --freeze "$H/corpus/fixtures/r97-clean"
echo "exit=$?"

echo
echo "=== verify clean copy ==="
echo "\$ python3 \$H/corpus/tools/dependency_freeze.py \$H/corpus/fixtures/r97-clean"
python3 "$H/corpus/tools/dependency_freeze.py" "$H/corpus/fixtures/r97-clean"
echo "exit=$?"

# mutation copies
rm -rf "$H/corpus/fixtures/r97-deps-reduction" "$H/corpus/fixtures/r97-key-row-deleted"
cp -r "$H/corpus/fixtures/r97-clean" "$H/corpus/fixtures/r97-deps-reduction"
cp -r "$H/corpus/fixtures/r97-clean" "$H/corpus/fixtures/r97-key-row-deleted"

python3 - "$H/corpus/fixtures/r97-deps-reduction" "$H/corpus/fixtures/r97-key-row-deleted" <<'PY'
import json, sys
from pathlib import Path

deps_dir, del_dir = (Path(p) for p in sys.argv[1:3])

p = deps_dir / "timeline.json"
data = json.loads(p.read_text(encoding="utf-8"))
before = list(data[-1]["depends_on"])
data[-1]["depends_on"] = []
p.write_text(json.dumps(data, indent=2) + "\n", encoding="utf-8")
print(f"deps-reduction: {data[-1]['id']} depends_on {before} -> []")

p = del_dir / "timeline.json"
data = json.loads(p.read_text(encoding="utf-8"))
removed = data.pop()
p.write_text(json.dumps(data, indent=2) + "\n", encoding="utf-8")
print(f"key-row-deleted: removed event {removed['id']}")

for d in (deps_dir, del_dir):
    if d is not del_dir:
        continue
    mp = d / "claim-excerpt-map.md"
    lines = mp.read_text(encoding="utf-8").splitlines()
    lines = [line for line in lines if not line.startswith("| J17 ")]
    for i, line in enumerate(lines):
        if line.startswith("**Expected inventory:**"):
            lines[i] = line.replace("17 JSON facts", "16 JSON facts").replace(
                "19 total mapped rows", "18 total mapped rows").replace(
                "15 distinct excerpt anchors", "14 distinct excerpt anchors")
            print(f"{d.name}: {lines[i]}")
    mp.write_text("\n".join(lines) + "\n", encoding="utf-8")
PY

echo
echo "=== verify deps-reduction copy (self-consistent maps) ==="
echo "\$ python3 \$H/corpus/tools/dependency_freeze.py \$H/corpus/fixtures/r97-deps-reduction"
python3 "$H/corpus/tools/dependency_freeze.py" "$H/corpus/fixtures/r97-deps-reduction"
echo "exit=$?"

echo
echo "=== verify key-row-deleted copy (self-consistent maps) ==="
echo "\$ python3 \$H/corpus/tools/dependency_freeze.py \$H/corpus/fixtures/r97-key-row-deleted"
python3 "$H/corpus/tools/dependency_freeze.py" "$H/corpus/fixtures/r97-key-row-deleted"
echo "exit=$?"

echo
echo "=== delivered tool on a FOREIGN-path copy (path binding) ==="
echo "\$ python3 \$REPO/corpus/tools/dependency_freeze.py \$H/corpus/fixtures/r97-clean"
python3 "$REPO/corpus/tools/dependency_freeze.py" "$H/corpus/fixtures/r97-clean"
echo "exit=$?"
