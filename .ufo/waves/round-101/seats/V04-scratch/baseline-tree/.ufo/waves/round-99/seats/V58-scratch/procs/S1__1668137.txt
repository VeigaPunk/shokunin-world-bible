/bin/bash
-c
cd '/home/vgpnk/Projects/shokunin' && set -u
base="/home/vgpnk/Projects/shokunin/lore-shokunin"
seats="$base/.ufo/waves/round-99/seats"
py="$base/corpus/tools/dependency_freeze.py"
contract="$base/corpus/tools/dependency-freeze-contract.json"
# capture baseline mtimes so we can detect S1 rewrite
py_m=$(stat -c '%Y' "$py")
ct_m=$(stat -c '%Y' "$contract")
echo "BASELINE py_mtime=$py_m ct_mtime=$ct_m"
echo "POLL_START $(date -Is)"
found=""
for i in $(seq 1 40); do
  hits=$(find "$seats" -maxdepth 2 -type f -name 'S1*' 2>/dev/null | head -20)
  # also any report mentioning freeze in seats root
  reports=$(find "$seats" -maxdepth 1 -type f 2>/dev/null | head -40)
  py_now=$(stat -c '%Y' "$py" 2>/dev/null || echo missing)
  ct_now=$(stat -c '%Y' "$contract" 2>/dev/null || echo missing)
  changed="no"
  if [ "$py_now" != "$py_m" ] || [ "$ct_now" != "$ct_m" ]; then changed="YES"; fi
  echo "POLL_CHECK try=$i t=$((i*20))s $(date -Is) s1_files=${hits:-none} py_changed=$([ "$py_now" != "$py_m" ] && echo yes || echo no) ct_changed=$([ "$ct_now" != "$ct_m" ] && echo yes || echo no) seat_reports=$(echo "$reports" | wc -l)"
  if [ -n "$hits" ]; then
    echo "POLL_HIT s1=$hits"
    found=1
    break
  fi
  # also accept freeze rewrite as artifact signal only if S1 report also appears; keep waiting for report
  sleep 20
done
if [ -z "$found" ]; then
  echo "POLL_TERMINAL status=ABSENT tries=40 elapsed=800s"
  echo "--- seats listing ---"
  find "$seats" -maxdepth 2 \( -type f -o -type d \) | head -200
  echo "--- tool mtimes final ---"
  stat -c '%y %n' "$py" "$contract"
fi
echo "POLL_END $(date -Is)"
