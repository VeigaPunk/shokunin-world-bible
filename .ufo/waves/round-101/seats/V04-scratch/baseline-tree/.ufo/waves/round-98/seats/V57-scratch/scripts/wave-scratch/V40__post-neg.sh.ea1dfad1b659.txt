#!/bin/bash
# V40 post-state negative matrix (C1): mismatch / migration / guard / path / tamper.
# Usage: post-neg.sh <mirror-root>   (mirror built by build-mirror.sh from delivered bytes)
set -u
M="$1"
TOOL="$M/corpus/tools/dependency_freeze.py"
CT="$M/corpus/tools/dependency-freeze-contract.json"
BASE="$M/base-contract.json"
R95="$M/corpus/fixtures/r95-crossing-aftermath"
R97="$M/corpus/fixtures/r97-crossing-acknowledgment"
R98="$M/corpus/fixtures/r98-clerk-filing"
run() { python3 "$TOOL" "$@"; echo "EXIT=$?"; }

echo "### P1a: SELF-TEST-MISMATCH exit-2 (clean.exit:=1)"
cp "$BASE" "$CT"
python3 - "$CT" <<'PY'
import json,sys
p=sys.argv[1]; c=json.load(open(p))
c["self_test"]["expected"]["clean"]["exit"]=1
json.dump(c,open(p,"w"),indent=2,ensure_ascii=False)
PY
run --self-test 2>&1 | tail -4

echo
echo "### P1b: SELF-TEST-MISMATCH exit-2 (digest_tamper.finding_ids:=[])"
cp "$BASE" "$CT"
python3 - "$CT" <<'PY'
import json,sys
p=sys.argv[1]; c=json.load(open(p))
c["self_test"]["expected"]["digest_tamper"]["finding_ids"]=[]
json.dump(c,open(p,"w"),indent=2,ensure_ascii=False)
PY
run --self-test 2>&1 | tail -4

echo
echo "### P1c: expected-set mismatch guard (delete clean key)"
cp "$BASE" "$CT"
python3 - "$CT" <<'PY'
import json,sys
p=sys.argv[1]; c=json.load(open(p))
del c["self_test"]["expected"]["clean"]
json.dump(c,open(p,"w"),indent=2,ensure_ascii=False)
PY
run --self-test 2>&1 | tail -3

echo
echo "### P2a: path-less legacy pin migrated by --freeze (canonical r95)"
cp "$BASE" "$CT"
python3 - "$CT" <<'PY'
import json,sys
p=sys.argv[1]; c=json.load(open(p))
c["pins"]["r95-crossing-aftermath"].pop("fixture",None)
json.dump(c,open(p,"w"),indent=2,ensure_ascii=False)
PY
run --freeze "$R95" 2>&1 | tail -5

echo
echo "### P2b: migrated pin payload vs base pin (minus revision)"
python3 - "$CT" "$BASE" <<'PY'
import json,sys
new=json.load(open(sys.argv[1]))["pins"]["r95-crossing-aftermath"]
old=json.load(open(sys.argv[2]))["pins"]["r95-crossing-aftermath"]
print("fixture key:",new.get("fixture"))
for k in ("content_sha256","files","events","relations","counts"):
    print(f"{k}: {'EQUAL' if new.get(k)==old.get(k) else 'DIFFERS'}")
print("revision:",old.get("revision"),"->",new.get("revision"))
PY

echo
echo "### P2c: verify canonical r95 after migration"
run "$R95" 2>&1 | tail -3

echo
echo "### P3a: --freeze out-of-scope target"
mkdir -p "$M/target"
run --freeze "$M/target" 2>&1 | tail -3

echo
echo "### P3b: --freeze in-scope foreign same-named path (pin WITH recorded path)"
cp "$BASE" "$CT"
mkdir -p "$M/corpus/fixtures/nested"
cp -r "$R95" "$M/corpus/fixtures/nested/r95-crossing-aftermath"
run --freeze "$M/corpus/fixtures/nested/r95-crossing-aftermath" 2>&1 | tail -3

echo
echo "### P3c: --freeze in-scope foreign same-named path (path-less legacy pin) [hole probe]"
cp "$BASE" "$CT"
python3 - "$CT" <<'PY'
import json,sys
p=sys.argv[1]; c=json.load(open(p))
c["pins"]["r95-crossing-aftermath"].pop("fixture",None)
json.dump(c,open(p,"w"),indent=2,ensure_ascii=False)
PY
run --freeze "$M/corpus/fixtures/nested/r95-crossing-aftermath" 2>&1 | tail -5
python3 -c "import json;print('pin fixture now:',json.load(open('$CT'))['pins']['r95-crossing-aftermath'].get('fixture'))"

echo
echo "### P4a: verify foreign same-named copy (pin WITH path) -> PATH-MISMATCH"
cp "$BASE" "$CT"
mkdir -p "$M/foreign"
cp -r "$R95" "$M/foreign/r95-crossing-aftermath"
run "$M/foreign/r95-crossing-aftermath" 2>&1 | tail -3

echo
echo "### P4b: verify canonical r95 -> PASS"
run "$R95" 2>&1 | tail -3

echo
echo "### P5a: cosmetic tamper r95 timeline.json -> DIGEST-MISMATCH"
cp "$BASE" "$CT"
python3 - "$M/corpus/fixtures/tamper-r95" "$R95" <<'PY'
import shutil,sys
shutil.copytree(sys.argv[2],sys.argv[1])
PY
python3 - "$M/corpus/fixtures/tamper-r95/timeline.json" <<'PY'
import sys
p=sys.argv[1]; t=open(p).read()
t=t.replace("Aru carries the north lantern","Aru carries the North lantern",1)
open(p,"w").write(t)
PY
python3 - "$CT" "$BASE" "$M/corpus/fixtures/tamper-r95" <<'PY'
import json,sys
ct=json.load(open(sys.argv[1])); base=json.load(open(sys.argv[2]))
pin=dict(base["pins"]["r95-crossing-aftermath"]); pin["fixture"]="corpus/fixtures/tamper-r95"
ct["pins"]["tamper-r95"]=pin
json.dump(ct,open(sys.argv[1],"w"),indent=2,ensure_ascii=False)
PY
run "$M/corpus/fixtures/tamper-r95" 2>&1 | tail -4

echo
echo "### P5b: semantic drop on r97 tick-13 edge -> REDUCTION-DETECTED"
cp "$BASE" "$CT"
python3 -c "import json,sys;sys.exit(0 if 'r97-crossing-acknowledgment' in json.load(open('$BASE'))['pins'] else 7)" || { echo "SKIP: r97 not in base contract"; }
mkdir -p "$M/corpus/fixtures/tamper-r97"
cp -r "$R97/." "$M/corpus/fixtures/tamper-r97/"
python3 - "$M/corpus/fixtures/tamper-r97/timeline.json" <<'PY'
import json,sys
p=sys.argv[1]; t=json.load(open(p))
for rec in t:
    if rec["id"]=="acknowledgment-aru-records":
        rec["depends_on"]=[d for d in rec["depends_on"] if d!="response-evin-salt-court"]
json.dump(t,open(p,"w"),indent=2)
PY
python3 - "$CT" "$BASE" "$M/corpus/fixtures/tamper-r97" <<'PY'
import json,sys
ct=json.load(open(sys.argv[1])); base=json.load(open(sys.argv[2]))
name="r97-crossing-acknowledgment"
pin=dict(base["pins"][name]); pin["fixture"]="corpus/fixtures/tamper-r97"
ct["pins"]["tamper-r97"]=pin
json.dump(ct,open(sys.argv[1],"w"),indent=2,ensure_ascii=False)
PY
run "$M/corpus/fixtures/tamper-r97" 2>&1 | tail -4

echo
echo "### P5c: semantic drop on r98 clerk-filing-open edge -> REDUCTION-DETECTED"
cp "$BASE" "$CT"
mkdir -p "$M/corpus/fixtures/tamper-r98"
cp -r "$R98/." "$M/corpus/fixtures/tamper-r98/"
python3 - "$M/corpus/fixtures/tamper-r98/timeline.json" <<'PY'
import json,sys
p=sys.argv[1]; t=json.load(open(p))
for rec in t:
    if rec["id"]=="clerk-filing-open":
        rec["depends_on"]=[]
json.dump(t,open(p,"w"),indent=2)
PY
python3 - "$CT" "$BASE" "$M/corpus/fixtures/tamper-r98" <<'PY'
import json,sys
ct=json.load(open(sys.argv[1])); base=json.load(open(sys.argv[2]))
name="r98-clerk-filing"
pin=dict(base["pins"][name]); pin["fixture"]="corpus/fixtures/tamper-r98"
ct["pins"]["tamper-r98"]=pin
json.dump(ct,open(sys.argv[1],"w"),indent=2,ensure_ascii=False)
PY
run "$M/corpus/fixtures/tamper-r98" 2>&1 | tail -4

echo
echo "### P6: delivered self-test full run"
cp "$BASE" "$CT"
run --self-test 2>&1

echo
echo "### P7: chain verify, every pinned fixture"
cp "$BASE" "$CT"
python3 - "$CT" <<'PY'
import json,sys
c=json.load(open(sys.argv[1]))
for name,pin in sorted(c["pins"].items()):
    print(f"{name}\t{pin['fixture']}")
PY
python3 - "$CT" "$M" <<'PY'
import json,subprocess,sys
c=json.load(open(sys.argv[1])); m=sys.argv[2]
for name,pin in sorted(c["pins"].items()):
    r=subprocess.run([sys.executable,m+"/corpus/tools/dependency_freeze.py",m+"/"+pin["fixture"]],
                     capture_output=True,text=True)
    print(f"CHAIN {name}: exit={r.returncode} {r.stdout.strip().splitlines()[-1] if r.stdout.strip() else ''}")
PY
