- B1 polling ran in one tool call every 60 seconds, with a 20-minute deadline. Each timestamped check read and parsed the JSON and recorded its SHA-256; it stopped after the same valid hash appeared in consecutive checks. Exact poll logic:

  ```sh
  python3 - <<'PY'
  import datetime, hashlib, json, pathlib, time
  p = pathlib.Path('/home/vgpnk/Projects/shokunin/lore-shokunin/.ufo/waves/round-90/patches/B1-tuples.json')
  deadline = time.monotonic() + 1200
  last_hash = None
  while True:
      now = datetime.datetime.now().astimezone().isoformat(timespec='seconds')
      try:
          raw = p.read_bytes()
          json.loads(raw)
          digest = hashlib.sha256(raw).hexdigest()
          stable = digest == last_hash
          print(f'{now} exists=true valid_json=true sha256={digest} stable_from_previous={stable} path={p}', flush=True)
      except FileNotFoundError:
          digest, stable = None, False
          print(f'{now} exists=false path={p}', flush=True)
      except Exception as exc:
          digest, stable = None, False
          print(f'{now} exists=true valid_json=false error={type(exc).__name__}:{exc} path={p}', flush=True)
      if digest is not None and stable:
          print(f'{now} stable_artifact=true', flush=True)
          break
      last_hash = digest
      if time.monotonic() >= deadline:
          print(f'{now} deadline_reached=true', flush=True)
          break
      time.sleep(60)
  PY
  ```
