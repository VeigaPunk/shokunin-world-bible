## Harness source (for reproducibility)

`census_i3.py`, run from the worktree root at `2b48ca7`; writes the two JSON artifacts and the
aggregate lines quoted under Method.

```python
import hashlib, json, re, subprocess, sys
from collections import Counter
from pathlib import Path

root = Path.cwd()
fixroot = root / 'corpus' / 'fixtures'
tuples, full = {}, {}
for d in sorted(p for p in fixroot.iterdir() if p.is_dir()):
    rel = f'corpus/fixtures/{d.name}'
    r = subprocess.run([sys.executable, 'corpus/fixtures/check_bible.py', rel],
                       cwd=root, capture_output=True, text=True)
    m = re.search(r'^(\d+) violation\(s\); no canon changed$', r.stdout, re.M)
    tuples[rel] = {'exit': r.returncode,
                   'stdout_sha256': hashlib.sha256(r.stdout.encode()).hexdigest(),
                   'violation_count': int(m.group(1)) if m else (0 if r.returncode == 0 else None)}
    full[rel] = {'command': f'python3 corpus/fixtures/check_bible.py {rel}',
                 'exit': r.returncode, 'stdout': r.stdout, 'stderr': r.stderr}
json.dump(tuples, open('r87-extended-tuples.json', 'w'), indent=2, sort_keys=True)
json.dump(full, open('I3-stdout.json', 'w'), indent=2, sort_keys=True)
ec = Counter(v['exit'] for v in tuples.values())
print('fixture_count', len(tuples))
print('exit_counts', dict(sorted(ec.items())))
print('total_violations', sum(v['violation_count'] or 0 for v in tuples.values()))
print('stderr_nonempty', sum(1 for v in full.values() if v['stderr']))
print('null_violation_count', [k for k, v in tuples.items() if v['violation_count'] is None])
```
