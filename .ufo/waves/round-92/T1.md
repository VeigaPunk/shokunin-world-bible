**Gate B baseline-only receipt replay.** This script parses fenced output blocks, pairs each with the preceding supported `sh`/`bash` block, expands literal fixture loops as data, invokes only the delivered checker with an allowlisted repo-relative fixture, and compares stdout bytes, return code, and empty stderr. It does not execute shell text from the document.

```sh
PYTHONDONTWRITEBYTECODE=1 python3 - <<'PY'
import pathlib, re, shlex, subprocess, sys
root=pathlib.Path.cwd(); doc=(root/'corpus/knowledge/holder-assessment-boundary.md').read_text()
fences=list(re.finditer(r'^```([^\n]*)\n(.*?)^```[ \t]*$',doc,re.M|re.S))
receipts=cases=0
for i,f in enumerate(fences):
    lang=f.group(1).strip().lower(); out=f.group(2)
    if lang in ('sh','bash') or not any(x in out for x in ('VIOLATION:','CLEAN: schema, citations, relations, timeline; 0 violations',' violation(s); no canon changed','=== ')):
        continue
    if i==0 or fences[i-1].group(1).strip().lower() not in ('sh','bash'):
        raise SystemExit(f'orphan receipt fence={i}')
    command=fences[i-1].group(2).replace('\\\n',' ')
    loop=re.search(r'for\s+(\w+)\s+in\s+([^;]+);\s*do(.*?)\bdone\s*$',command,re.S)
    if loop:
        var,names_text,body=loop.groups(); names=shlex.split(names_text.strip())
        m=re.search(r'python3\s+corpus/fixtures/check_bible\.py\s+([^\s;]+)',body)
        if not m or not re.search(r'echo\s+"?===\s*\$'+re.escape(var)+r'\s*==="?',body) or not re.search(r'echo\s+"?\[rc=\$\?\]"?',body):
            raise SystemExit(f'unsupported loop receipt={i}')
        targets=[(n,m.group(1).replace('"$'+var+'"',n).replace('$'+var,n).strip('"')) for n in names]
    else:
        m=re.fullmatch(r'\s*python3\s+corpus/fixtures/check_bible\.py\s+([^\s;]+)\s*',command)
        if not m: raise SystemExit(f'unsupported command receipt={i}')
        targets=[(None,m.group(1).strip('"'))]
    rows=list(re.finditer(r'^=== ([A-Za-z0-9_.-]+) ===\n(.*?)^\[rc=(\d+)\]$',out,re.M|re.S))
    if targets[0][0] is None:
        m=re.fullmatch(r'(.*?)\n\[rc=(\d+)\]\n?',out,re.S); rows=[m] if m else []
    if len(rows)!=len(targets): raise SystemExit(f'receipt row count mismatch fence={i}')
    ok=True
    for (name,target),row in zip(targets,rows):
        if name is None: expected_stdout,expected_rc=row.groups()
        else:
            found,expected_stdout,expected_rc=row.groups()
            if found!=name: ok=False
        p=pathlib.PurePosixPath(target)

        ok &= match; cases+=1
    if not ok: raise SystemExit(f'receipt mismatch fence={i}')
    receipts+=1
print(f'replay_receipts={receipts}/{sum(1 for i,f in enumerate(fences) if f.group(1).strip().lower() not in ("sh","bash") and any(x in f.group(2) for x in ("VIOLATION:","CLEAN: schema, citations, relations, timeline; 0 violations"," violation(s); no canon changed","=== ")))}')
```
