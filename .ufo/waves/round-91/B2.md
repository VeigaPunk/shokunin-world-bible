The command below is the exact path-limited archive/census invocation. It ran the archived checker once per direct fixture directory, with reverse-alphabetical iteration. `python3 -I` is used for both the census driver and every checker subprocess.

```sh
set -o pipefail
scratch=$(mktemp -d /tmp/lore-r91-B2-census.XXXXXX) || exit 1
trap 'rm -rf -- "$scratch"' EXIT
git archive --format=tar a7c6a1773d5994246a76e4eff30d645f161920e0 corpus/fixtures | tar -xf - -C "$scratch" || exit 1
printf 'archive_command=git archive --format=tar a7c6a1773d5994246a76e4eff30d645f161920e0 corpus/fixtures | tar -xf - -C %s\narchive_extract_exit=0\nscratch_root=%s\n' "$scratch" "$scratch"
mkdir -p .ufo/waves/round-91/patches || exit 1
B2_SCRATCH="$scratch" B2_OUTPUT="$PWD/.ufo/waves/round-91/patches/B2-tuples.json" python3 -I - <<'PY'
import hashlib, json, os, pathlib, subprocess
root = pathlib.Path(os.environ['B2_SCRATCH'])
fixtures = root / 'corpus/fixtures'
names = sorted((p.name for p in fixtures.iterdir() if p.is_dir()), reverse=True)
rows = {}
exit_counts = {0: 0, 1: 0}
stderr_nonempty = []
unexpected = []
for name in names:
    rel = f'corpus/fixtures/{name}'
    result = subprocess.run(['python3', '-I', 'corpus/fixtures/check_bible.py', rel], cwd=root, stdout=subprocess.PIPE, stderr=subprocess.PIPE)
    exit_counts[result.returncode] = exit_counts.get(result.returncode, 0) + 1
    if result.stderr:
        stderr_nonempty.append(rel)
    if result.returncode not in (0, 1):
        unexpected.append((rel, result.returncode))
    rows[rel] = {
        'exit': result.returncode,
        'stdout_sha256': hashlib.sha256(result.stdout).hexdigest(),
        'violation_count': sum(line.startswith(b'VIOLATION:') for line in result.stdout.splitlines()),
    }
output = json.dumps(rows, indent=2, sort_keys=True) + '\\n'
output_path = pathlib.Path(os.environ['B2_OUTPUT'])
output_path.write_text(output, encoding='utf-8')
raw = output.encode('utf-8')
print(f'fixture_count={len(names)}')
print(f'reverse_first={names[0]} reverse_last={names[-1]}')
print(f'exit_counts={exit_counts}')
print(f'total_violation_lines={sum(r["violation_count"] for r in rows.values())}')
print(f'stderr_nonempty_count={len(stderr_nonempty)}')
print(f'unexpected_exit_count={len(unexpected)}')
if unexpected:
    print(f'unexpected_exits={unexpected}')
print(f'tuple_file_bytes={len(raw)}')
print(f'tuple_file_sha256={hashlib.sha256(raw).hexdigest()}')
print(f'tuple_file={output_path}')
PY
census_status=$?
printf 'census_exit=%s\n' "$census_status"
exit "$census_status"
```

The round-91 B1 availability poll ran within one tool call. Exact command:

```sh
set -o pipefail
python3 - <<'PY'
import datetime, hashlib, json, pathlib, time
p = pathlib.Path('/home/vgpnk/Projects/shokunin/lore-shokunin/.ufo/waves/round-91/patches/B1-tuples.json')
deadline = time.monotonic() + 1200
last_hash = None
terminal = False
def interrupted(signum, frame):
    global terminal
    now = datetime.datetime.now().astimezone().isoformat(timespec='seconds')
    print(f'{now} terminal=interrupted signal={signum}', flush=True)
    terminal = True
    raise SystemExit(128 + signum)
import signal
signal.signal(signal.SIGINT, interrupted)
signal.signal(signal.SIGTERM, interrupted)
while True:
    now = datetime.datetime.now().astimezone().isoformat(timespec='seconds')
    try:
        raw = p.read_bytes()
        json.loads(raw)
        digest = hashlib.sha256(raw).hexdigest()
        stable = digest == last_hash
        print(f'{now} exists=true valid_json=true sha256={digest} stable_from_previous={stable} path={p}', flush=True)
    except FileNotFoundError:
        digest, stable = None, False
        print(f'{now} exists=false path={p}', flush=True)
    except Exception as exc:
        digest, stable = None, False
        print(f'{now} exists=true valid_json=false error={type(exc).__name__}:{exc} path={p}', flush=True)
    if digest is not None and stable:
        print(f'{now} stable_artifact=true', flush=True)
        print(f'{now} terminal=stable_artifact', flush=True)
        break
    last_hash = digest
    if time.monotonic() >= deadline:
        now = datetime.datetime.now().astimezone().isoformat(timespec='seconds')
        print(f'{now} deadline_reached=true', flush=True)
        print(f'{now} terminal=deadline', flush=True)
        break
    time.sleep(60)
PY
```
