Exact verifier command and code (it executes the pinned source through `python -c` and the delivered checker through its CLI entry point):

```sh
python3 - /tmp/lore-r91-X1-verify /home/vgpnk/Projects/shokunin/lore-shokunin/corpus/fixtures/check_bible.py <<'PY'
import difflib, pathlib, re, subprocess, sys
scratch, checker = pathlib.Path(sys.argv[1]), pathlib.Path(sys.argv[2])
repo = pathlib.Path('/home/vgpnk/Projects/shokunin/lore-shokunin')
revision = 'e97613cc276f5fa9cac49ff1900798cf6863c020'
old_source = subprocess.run(['git', 'show', f'{revision}:corpus/fixtures/check_bible.py'], cwd=repo, capture_output=True, text=True, check=True).stdout
wrapper = 'import sys; source="/home/vgpnk/Projects/shokunin/lore-shokunin/corpus/fixtures/check_bible.py"; code=sys.stdin.read(); sys.argv=[source, *sys.argv[1:]]; exec(compile(code, source, "exec"), {"__name__":"__main__", "__file__":source})'
fixtures = ('r91-c1-null-chain', 'r91-c1-tick-unresolved')
for name in fixtures:
    directory = scratch / 'corpus/fixtures' / name
    readme = (directory / 'README.md').read_text(encoding='utf-8')
    expected = dict(re.findall(r'## Expected output with (e97613c|the delivered checker) \(exit 1; stderr empty\)\n\nComplete stdout:\n\n```text\n(.*?)\n```', readme, re.S))
    assert set(expected) == {'e97613c', 'the delivered checker'}, (name, expected.keys())
    before = subprocess.run([sys.executable, '-c', wrapper, str(directory)], cwd=repo, input=old_source, capture_output=True, text=True)
    after = subprocess.run([sys.executable, str(checker), str(directory)], cwd=repo, capture_output=True, text=True)
    assert before.returncode == after.returncode == 1, (name, before.returncode, after.returncode)
    assert before.stderr == after.stderr == '', (name, before.stderr, after.stderr)
    assert before.stdout == expected['e97613c'] + '\n', (name, 'e97613c output differs from README')
    assert after.stdout == expected['the delivered checker'] + '\n', (name, 'delivered output differs from README')
    removed = [line[2:] for line in difflib.ndiff(before.stdout.splitlines(), after.stdout.splitlines()) if line.startswith('- ')]
    added = [line[2:] for line in difflib.ndiff(before.stdout.splitlines(), after.stdout.splitlines()) if line.startswith('+ ')]
    assert len(removed) == len(added) == 1, (name, removed, added)
    if name == 'r91-c1-null-chain':

        chain_before = [line for line in before.stdout.splitlines() if 'possible provenance chain break' in line]
        chain_after = [line for line in after.stdout.splitlines() if 'possible provenance chain break' in line]
        assert chain_before == chain_after and len(chain_after) == 1
    print(f'{name}: e97613c_exit=1 delivered_exit=1 stderr=empty README_exact=YES changed_findings=1 outputs_differ=YES')
PY
```
